% we only need the moment about the mean
% so we define a special version of skew

procedure momentM(X,r);
begin
  scalar n, mom;
  n   := length(X);  
  me  := (for each el in X  sum el) / length(X);
  mom := for each el in X  sum (el - me)^r;
  return mom/n
end;

procedure skew(X);
    momentM(X,3)/ sqrt(momentM(X,2)^3); %--- compatibel with MatLab
    
% ------------------- Example usage ---------------------

on rounded $
print_precision 5 $

X := {4, 36, 45, 50, 75}$ 

skew(X);    %  = -0.306  MatLAB
