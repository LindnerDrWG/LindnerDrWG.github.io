%--- b.

% FIRST version of binoINV using a bisection method:

procedure binoINV(q,n,p);
begin scalar a, b, m;
      a:=0; b:=n;
      while (b-a>1) do 
         begin
            m := 0.5*(a+b);
            if binoCDF(m,n,p) < q then a:=m else b:=m;
         end;
      return  floor(b)
end;
 

binoINV(0.7, 100, 0.3);
