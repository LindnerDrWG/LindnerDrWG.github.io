procedure chi2CDF(x, k);
          if x<=0 or k<=0 then 0.0 
          else iGamma(k*0.5, x*0.5) $

% ------------------- Example usage ---------------------

on rounded $
print_precision 5$

X := {1,2,4,8,16,32};
write("x   chi2CDF(x)   P-value")$
for each el in X do write {el,  chi2CDF(el,3), 1-chi2CDF(el,3)} ;
