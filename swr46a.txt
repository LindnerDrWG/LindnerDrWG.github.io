% ------------------- helper functions ---------------------
procedure chi2CDF(x, k);
          if x<=0 or k<=0 then 0.0 
          else iGamma(k*0.5, x*0.5) $
procedure chi2inv(p,nu);  % use only chi2CDF
  begin scalar lo,hi,mid,fmid, tol:= 1.0e-10;
    lo := 0.0; hi := nu;
    while chi2CDF(hi,nu) < p do hi := 2.0*hi;
    for i := 1:1000 do
    << mid := (lo+hi)/2.0;   % bisection
       fmid := chi2cdf(mid,nu);
       if abs(fmid-p) < tol then  mid; 
       if fmid < p then lo := mid else hi := mid;
    >>;   return (lo+hi)/2.0;
  end;
procedure SD(lst);
   begin scalar n, mu, ds;
     n := length(lst);  
     mu := (for each el in lst sum el) / n;
     ds := for each el in lst sum (el - mu)^2;
     return sqrt(ds/(n-1))       % with n-1, sample
   end;
procedure MEAN(X);  (for each el in X sum el)/length(X);
procedure SUM_(X);  (for each el in X sum el);
define    DIM = length$


%#################################################
%      EXAMPLE: The size of PREY
%#################################################*/
%      We first do the tDIFF STEP-BY-STEP

on rounded $
print_precision 5$

X := {12.43,11.71,14.41,11.05,9.53, 11.66, 9.33, 11.71, 14.35, 13.81}$

%-- hypothesized standard deviation 
sigma0 := 1 $  

%-- significance level of rejecting the null hypothesis:
alpha  := 0.05  $ 

n := dim(X) $
s := sd(X) ;
df := n-1 $                        %-- degrees of freedom 

T_ := (n-1)*s*s/(sigma0*sigma0);    %-- test statistic 

pValue := min( 2*chi2CDF(T_,df), 2*(1-chi2CDF(T_,df)) ) ;

/*--  find Chi-Square critical value */
critValue := chi2INV(1-0.05, df);    %-- 'lower.tail=FALSE' ; .o5 illegal

COMMENT  Note
-- lower.tail: If TRUE, the probability to the left of p in the F 
-- distribution is returned. CAS R:  Default is TRUE.
-- If FALSE, the probability to the right is returned. 
-- If the test statistic T is greater than 16.9189, then the results 
-- of the test are statistically significant.
-- Note that smaller values of alpha will lead to larger Chi-Square 
-- critical values. $

if (pValue < 0.975) then write("accept H0") else write("reject H0");

ucl := (n-1)*s^2/ chi2INV(alpha/2, df) ;
lcl := (n-1)*s^2/ chi2INV(1-alpha/2, df) ;
write {lcl, ucl};     %-- confidence intervall 


COMMENT ---------------------------------------- Result

T         :   29.208
pValue    :   0.0011956
critValue : 16.919
CI        :  [1.5354, 10.816]
accept H0
