%% according to  math formula
procedure binoCDF1(k,n,p);
          for j := 0:k sum binomial(n,j)*p^j*(1-p)^(n-j);

%% according to remark
procedure binoCDF(k,n,p);
          iBeta(n-k,k+1,1-p);

% -------------------- Example usage ---------------------

% if switch rounded is off or not activated
% binoCDF is 'only' evaluated symbolically,
% i.e. iBeta does not return a number:

binoCDF(4, 10, 0.5);       % no number back


% if switch rounded is on i.e. numerically
% evaluation is activated, then we get
% a numeric result as a floating point number:

on rounded $
print_precision  5$

binoCDF(4, 10, 0.5);    % 0.3769
binoCDF1(4, 10, 0.5);   % 0.3769
