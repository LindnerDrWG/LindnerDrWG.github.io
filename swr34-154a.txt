procedure fPDF(x,f1,f2);
          sqrt(((f1*x)^f1*f2^f2)/(f1*x+f2)^(f1+f2)) / (x * beta(f1/2,f2/2)) $

procedure fCDF(x,df1,df2);
          iBeta( 0.5*df1, 0.5*df2, x/(x+df2/df1))$


% ---------- Newton-Raphson Solver for Inverse F ----------
%  Newton-Raphson Solver for Inverse F Distribution 
% found by Google GEMINI with question
% 'Inverse F Distribution via Newton Iteration in REDUCE 3.8'

procedure fINV_Newton(p, df1, df2, x0, max_iter);
   begin
      scalar x, x_new, val, deriv;
      on rounded;
      x := x0;
      write "Starting Newton Iteration for Inverse F Distribution...";
      write "Target Probability p = ", p, ", df1 = ", df1, ", df2 = ", df2;
      for i := 1:max_iter do begin
         val   := fcdf(x, df1, df2) - p;
         deriv := fpdf(x, df1, df2);
         if abs(deriv) < 1e-12 then begin
            write "Error: Derivative too close to zero.";
            return x;
         end;
         x_new := x - val / deriv;
         write "Iteration ", i, ": x = ", x_new, ", residual = ", val;
         if abs(x_new - x) < 1e-7 then begin
            write "Convergence achieved successfully.";
            return x_new
            end;
         x := x_new;
      end;
      write "Warning: Maximum iterations reached without full convergence.";
      return x
   end;

% ------------------- Example usage ---------------------

% Find the critical value for p = 0.95, df1 = 5, df2 = 10, 
%      starting from guess x0 = 2.0

on rounded $
print_precision 5$

scalar result;
result := fINV_Newton(0.95, 5, 10, 2.0, 20);
write "Final Inverse F Value: ", result;
