% Because REDUCE is case-insensitive we use the identifier
% NN instead of N etc.
% So the parameter list reads
%               N  R  n r

procedure hgPDF(NN,RR,n,r); 
          binomial(RR,r)*binomial(NN-RR, n-r)/binomial(NN,n)$

% -------------------- Example usage ---------------------

on rounded;
print_precision 5;

hgPDF(100,70,20,14);
