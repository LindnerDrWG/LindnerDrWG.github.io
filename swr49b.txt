% ------------------- helper functions ---------------------
% -- These problems are dominated by the binomial distribution.
% -- To calculate the sign-TEST score we need the function
%   -- binoPDF() as helper:

precision 10 $
procedure binoPDF(x,n,p);
          binomial(n,x)*p^x*(1-p)^(n-x) $ % x=0,1,2,...n
procedure count_gr(X,a); 
   begin scalar s;
     for each el in X do if (el>a) then s:=s+1;
     return s
   end;
procedure count_neq(X,a); 
   begin scalar s;
     for each el in X do if not(el=a) then s:=s+1;
     return s
   end;
procedure count_eq(X,a); 
   begin scalar s;
      for each el in X do if (el=a) then s:=s+1;
      return s
   end;
define    DIM = length$

% ---------------------------------------------------
procedure  signtest2(X, Y,  method);     
%------------ method: left=-1, right=1,both=0 -------

begin scalar n,p,p1,p2,s,nt, D;
  n  := dim(X);
  D  := for i:=1:n collect part(X,i)- part(Y,i); % D = X-Y
  nt := count_neq(D,0);
  s  := count_gr(D,0);
  if (method = 0) then % both
      << p1 := for i:=0:s sum binoPDF(i,nt,0.5);
         p2 := 1 - (for i:=0:(s-1) sum binoPDF(i,nt,0.5) );
         p  := min(2 * p1, 2 * p2) >>;
  if (method = 1) then % right
        p := 1 - (for i:=0:(s-1) sum binoPDF(i, nt, 0.5) );
  if (method = -1) then  % left
         p := for i:=0:s sum binoPDF(i, nt, 0.5);
   write "s= ",s;
   write "p_= ",p;
  % mat((s_, pp),(s,p));
END; %...............................................
