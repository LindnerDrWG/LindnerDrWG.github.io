procedure paretoPDF(x, m, a);
          if x<m then 0.0 else a * m^a / x^(a+1) $


% ------------------- Example usage ---------------------

on rounded $
print_precision 5$


%----- a.

shape := {1,2,3,4,5}$
for each y in shape do write {y, paretoPDF(3,2,y)} ;
