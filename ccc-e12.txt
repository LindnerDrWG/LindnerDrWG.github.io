load_package excalc;

% 1. Define a 3D coordinate frame to register indices (x,y,z) aka (1,2,3)
coframe e(1)=d x, e(2)=d y, e(3)=d z;

% 2. Define a core Reduce matrix A and fill it with symbolic variables
matrix a(3,3);
a(1,1):=a11; a(1,2):=a12; a(1,3):=a13;
a(2,1):=a21; a(2,2):=a22; a(2,3):=a23;
a(3,1):=a31; a(3,2):=a32; a(3,3):=a33;
a;

% 3. Calculate determinant expansion using eps
det_expansion := for each i in {1,2,3} sum
                 for each j in {1,2,3} sum
                 for each k in {1,2,3} sum
                     eps(i,j,k) * a(1,i) * a(2,j) * a(3,k);

% 4. Define a matrix and fill it with concrete values
matrix M(3,3);
M(1,1):=2; M(1,2):=2; M(1,3):=4;
M(2,1):=1; M(2,2):=2; M(2,3):=-5;
M(3,1):=3; M(3,2):=1; M(3,3):=-3;
M;

det_expansion := for each i in {1,2,3} sum
                 for each j in {1,2,3} sum
                 for each k in {1,2,3} sum
                   eps(i,j,k) * m(1,i) * m(2,j) * m(3,k);


end;
