load_package excalc;

pform {x,y,z, fx,fy,fz, c,c1}=0, w=2;

% component functions of vector field F
fx := x*y**2*z;  
fy := -2*x**3;  
fz := y*z**2;

% define surface constrained, so d z becomes -2x d x - 2y d y
z := 1 - x**2 - y**2;      

% define vector field F as 2-form w  and automatically
% evaluate the 2-form on the surface constraint            
        
w := fx * d y^d z + fy * d z^d x + fz * d x^d y;

% capture coefficient of d x^d y with inner product
c := @(y) _| (@(x) _| w);   

% convert to polar coordinates (dx^dy = r*dr^dth)
% and integrate with double integral over the unit disk x^2 + y^2 <= 1 

c1 := r * sub( x=r*cos(th), y=r*sin(th), c);
int(int(c1, th, 0, 2*pi), r, 0, 1);


end;
