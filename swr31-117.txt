% ---------- inverse erf and normINV -----------------
% this code is from Gemini AI with minor corrections.
% It implements a Newton-Raphson algorithm.

% load_package specfn;   -- erf is autoloaded

procedure erfinv(x);
begin
  scalar y, old_y, sqrt_pi, precision_threshold;
  y := x * 0.8862;          
  % Initial guess close to x * sqrt(pi)/2
  precision_threshold := 1.0e-12;
  sqrt_pi := sqrt(pi);
  repeat 
     << old_y := y;
        y := old_y-(erf(old_y)-x)/((2/sqrt_pi)*exp(-old_y^2));
     >> 
    until abs(y - old_y) < precision_threshold;
  return y
end;

procedure normINV(q,m,s);
   begin  if q=0 then return "-Inf";
          if q=1 then return "+Inf";
          if 0<q and q<1 then 
             return m + sqrt(2) * s * erfinv(2*q-1)
    end$

procedure stdnormINV(q); normINV(q,0,1) $

% ------------------- Example usage ---------------------

stdnormINV(1/10);
stdnormINV(2/10);
