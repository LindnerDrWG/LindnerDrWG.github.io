procedure weibullCDF(x,a,b);
          if (x<0) then 0.0 else  1 - exp(-(x/a)^b) $

% ------------------- Example usage ---------------------

on rounded $
print_precision 5$

shape := {0.5,1,1.5,2,2.5,3,3.5,4,4.5,5}$

for each y in shape do write {y, weibullCDF(1,1,y)} ;

% the value for (1,1,5) is verified on the green graph.
