procedure poissonPDF(k, lam);
          exp(-lam) * lam^k / factorial(k);

procedure poissonCDF(x, lam);
          for k:=0:floor(x)  sum poissonPDF(k, lam);


% -------------------- Example usage ---------------------

on rounded$
print_precision 5$


1 - poissonCDF(4,2);
