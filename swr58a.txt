procedure randlist(n); for i:=1:n collect rnd() $
procedure sum_(lst);   for each el in lst sum el $ 
procedure mean(lst);   sum_(lst)/length(lst) $ 
procedure listmake(m); for i:=1:m collect 0$
procedure SD(lst);
begin scalar n, mu, ds, n:=length(lst);
  mu := (for each el in lst sum el)/n;  % mean
  ds := for each el in lst sum (el - mu)^2;
  return sqrt(ds/n)
end$

procedure listmake(m); for i:=1:m collect 0$

procedure putNTH(L,k,v);
 %    putNTH sets k-th element of list L to value v
 %    because part(L,k) gives only _reading_ access
 begin scalar a:=L, b:={};;
    for i:=1:(k-1) do
         << b:=append(b,{first a}); a:=rest a>>;
    b := append(b,{v}); a := rest a;
    return append(b,a)
 end;         %  putnth({1,2,3},2,9) --> {1,9,3};

print_precision $

X := {2,1,3,4,7}$
n := dim(X);                     %-- number of samples 
IT := 10;                        %-- number of iterations 

M := listmake(IT);               %-- container for all samples

S := listmake(IT);               %-- container for all samples

Mboot := listmake(n);            %-- container for all means

Sboot := listmake(n);            %-- container for all sd's

for j:=1:IT do  
    begin                         %-- start bootstrap samples 
      for i:=1:n do               %-- create one resample of X
          begin r := 1 + random(n);  
            putNTH( Mboot,i, part(X,r));
            putNTH( Sboot,i, part(X,r))
           end;                    %-- bootstrap new sample 
       putNTH( M,j,  Mean(Mboot)); %-- j.th bootstrap mean 
       putNTH( S,j,  sd(Sboot));   %-- j.th bootstrap sd 
       write M(j)," ; ",S(j)       %-- show intermediate results
     end;                     

estM := Mean(M);                   %-- bootstrap sample mean/sd

estS := sd(S);                     %-- i.e. estimated values */ ;

load_package matrix$

mat( (estMean_, estSD_),(estM, estS));
