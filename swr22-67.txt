procedure geoCDF(n,p);
          1-(1-p)^(n+1) ;             % -- explicit formula

procedure geoCDF1(n,p);
         for k:=0:n sum geoPDF(k,p);  % -- math. definition

% -------------------- Example usage ---------------------

on rounded $
print_precision 5$

geoCDF(8,0.6);        
geoCDF1(8,0.6);
