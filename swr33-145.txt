procedure tPDF(x,f);  
          1/(sqrt(f)*Beta(1/2,f/2))*(1+ x^2/f)^(-0.5*(f+1)) $

procedure tCDF(a,df);
   if a=0 then 0.5 else 
   if a<0 then 0.5 * ibeta(df/2, 0.5, df/(a^2+df)) else
           1 - 0.5 * ibeta(df/2, 0.5, df/(a^2+df));

% ---------- a. bisection method

% Inverse Student-t distribution using bisection method
procedure tinv1(p,nu);
   begin scalar lo:=0, hi:=100, eps:= 1.0e-10,
                mid,flo,fmid;
      while hi-lo > eps do
         begin mid  := (lo+hi)/2;
               fmid := tcdf(mid,nu);
               if fmid < p then lo:=mid else hi:=mid;
         end;
      return (lo+hi)/2;
   end;


% ---------- b. NEWTON iteration by ChatGPD

procedure tinv2(p,nu);
   begin scalar x, xnew, f, fx, n,
      tol:= 1.0e-10, maxit:= 100;
      x := 0; % Initial approximation
              % x := tan(pi*(p-0.5)); would be a  
              % Cauchy-based starting guess, exact when nu=1
      % Newton iteration
      for n := 1:maxit do
        begin
          f  := tcdf(x,nu) - p;  % nominator
          fx := tpdf(x,nu);      % denominator
          xnew := x - f/fx;
          if abs(xnew-x) < tol then return xnew;
          x := xnew;
        end;
      return x
   end;

% -------------- c. version: NEWTON iteration by fjW -----------

procedure tinv2a(p,nu);     % enhanced by F.J. Wright
   begin scalar x, f, fx, dx,
      tol:= 1.0e-8, maxit:= 100;
      x := 0;  
      repeat <<
         f  := tcdf(x,nu) - p;
         fx := tpdf(x,nu);
         dx := f/fx;
         x := x - dx;
         maxit := maxit - 1;
      >> until abs dx < tol or maxit = 0;
      return x
   end;

% -------------- d. tiNV via num_solve  -----------------

procedure tINV(p,df); 
      <<
       clear x;
       if x=0.0 then -infinity else
       if x=1.0 then +infinity else
       rhs first num_solve(1-0.5*ibeta(df/2,0.5,df/(x^2+df))=p, x=2)
      >>$
