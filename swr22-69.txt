procedure geoINV(u,p);
          ceiling(( log(1-u)/log(1-p) - 1)) $


%   Function: ceiling (x)
%   When x is a real number, return the least integer 
%        that is greater than or equal to x. 
%   Function: log (x)
%   represents the _natural_ base logarithm of x.

% -------------------- Example usage ---------------------

on rounded $
print_precision 5$

geoINV(0.70, 1/6);
geoINV(0.70, 0.4);
