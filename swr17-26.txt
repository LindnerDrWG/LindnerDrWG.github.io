load_package assist;

procedure  median(X);
begin
  scalar  Xs,n2,n2p; % local vars
          Xs   := sortnumlist X; 
          n2   := floor(length(X)/2);  
          n2p  := n2+1;
          if evenp length(X) 
             then return 0.5*(part(Xs,n2)+part(Xs,n2p))
             else return  part(Xs,n2p)
end;

% ------------------- Example usage ---------------------

on rounded $
print_precision 5$

X := {4, 36, 45, 50, 75}$
median(X);
