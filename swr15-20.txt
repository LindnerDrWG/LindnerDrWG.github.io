procedure mad(lst);
begin
  scalar n, mu, diff_sum;
  n  := length(lst);  
  mu := (for each el in lst sum el)/n;  % mean
  diff_sum := for each el in lst sum abs(el - mu);
  return diff_sum/n
end;

procedure cumsum(lst);
  begin scalar cumlist;
    cumlist := {};
    for i:=1:length(X) do 
        cumlist := append(cumlist, {for j:=1:i sum part(X,j)});
    return cumlist
end;

procedure  median(X);
begin
  scalar  Xs,n2,n2p; % local vars
          Xs   := sortnumlist X; 
          n2   := floor(length(X)/2);  
          n2p  := n2+1;
          if evenp length(X) 
             then return 0.5*(part(Xs,n2)+part(Xs,n2p))
             else return  part(Xs,n2p); 
end;
