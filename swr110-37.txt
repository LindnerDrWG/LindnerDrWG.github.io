           STATISTICS WITH REDUCE ch. 1.10-37
                      by W.G. Lindner 2026  $
                      

                      
procedure moment(X,A,r);
         (for each el in X  sum (el - A)^r )/length(X);

% ------------------- Example usage ---------------------

on rounded $
print_precision 5$

X := { 1 }$

moment(X,A,3);
moment(X,0,3);
