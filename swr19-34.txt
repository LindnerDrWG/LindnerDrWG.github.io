%-- We use the 'nearest-rank method' for calculating the quantile.

procedure quantile(X,p);              
begin scalar Xs, i, dim;
      if p >= 100 then return max(X);
      if p <= 0 then return min(X);
      dim := length(X);
      Xs  := sort(X, <);
      i   := 1 + round((dim-1)*p/100);
      return part(Xs, i)
end;

define percentile = quantile;  %-- alias for Matlab compatibility
define Qp =         quantile;  %-- relax notation


%-- Define the 5-value-summary (Tukey) and algebraic.boxplot().

procedure Q0(X) ; quantile(X,0);      %--   0% quartile
procedure Q1(X) ; quantile(X,25);     %--  25% quartile, i.e. lower Q
procedure Q2(X) ; quantile(X,50);     %--  50% quartile, i.e. median
procedure Q3(X) ; quantile(X,75);     %--  25% quartile, i.e. upper Q
procedure Q4(X) ; quantile(X,100);    %-- 100% quartile

procedure boxplot(X);
     mat(( p0, p25 , p50, p75, p100),
         (Q0(X), Q1(X), Q2(X), Q3(X), Q4(X)));

% ------------------- Example usage ---------------------

df := {6,3,2,10,1};
on rounded $
print_precision 5$

boxplot(df);
quantile(df, .4);

  
end;
