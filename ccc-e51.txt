load_package listvecops;

listproc cross(a, b);  {(a _2)*(b _3) - (a _3)*(b _2), 
                        (a _3)*(b _1) - (a _1)*(b _3), 
                        (a _1)*(b _2) - (a _2)*(b _1)};

cross({1,2,3}, {4,5,6});

listproc curl(F);  {df(F _3,y) - df(F _2,z), 
                    df(F _1,z) - df(F _3,x), 
                    df(F _2,x) - df(F _1,y)};

curl({x^2,y^2,1});    % = {0,0,0} ok.

F:= {-y^3,x^3,-z^3}; 
curl(F);              % = {0,0, 3x^2+3y^2}


listproc normalize v;         v / sqrt(v *. v);
define normal = cross;
listproc unit_normal (a,b);   normalize( normal(a,b) );
      
% vertices
A := {0,0,0};  B := {0,2,0};  C := {0,0,2};
N := unit_normal (B-A,C-A);   


end;
