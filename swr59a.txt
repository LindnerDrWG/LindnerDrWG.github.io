COMMENT EXAMPLE from AskPython
-- To understand the need for bootstrap sampling let’s consider an 
-- example, suppose we want to calculate the average age of 1000 
-- employees working for a particular company. 
-- The first approach could be to ask all the 1000 employees their  
-- age and then calculate the mean age.
-- This can be a tedious task and a time-consuming approach, 
-- another method is to consider a sample of 5 employees and collect 
-- their ages. This process can be repeated 20 times, and
-- then average the collected age data of 100 employees. 
-- This average age would be the estimate of all 1000 employees. $


%-- FIRST We follow the example step by step.
%-- a. - calculate the estimated population mean age

on rounded $
print_precision $

X := {25, 30, 35, 40, 45, 50, 55, 60, 65, 70}$

n  := dim(X);
numSamples := 3;   %-- choose number of bootstrap samples

%-- get random index r to resample with replacement from the 
%-- original sample to generate a bootstrap sample.

r := 1 + random(10);

%-- create a list to save the bootstrap results of 10 drawings
M := listmake(n);

%-- create a list as container to save the means of the drawings
bootstrapMeans := listmake(numSamples);

%-- now resample with replacement from the original sample X to
%-- generate a bootstrap sample in M:

for j:=1:numSamples do  %-- perform bootstrap sampling
     begin           
     for  i:=1:n do 
            begin r := 1 + random(10);
              M := putNTH( M, i, part(X,r));
              write M;
              % Now calculate the mean age of the 
              % bootstrap sample M; this calculated mean is 
              % stored in list bootstrapMeans:
              bootstrapMeans := putNTH( bootstrapMeans, j, Mean(M));
              write bootstrapMeans
            end;
end;

%-- Now we calculate the estimated population mean age
estimatedMean := Mean(bootstrapMeans);

%- c. -  Bootstrap Sampling for Confidence Interval (CI).
%  -- calc confidence intervals via percentiles for the
%  -- bootstrapMeans

bootstrapMeans := {47, 50, 51.5};   %-- fix bsMeans from above

ci1 := percentile(bootstrapMeans, 98.5);   
ci2 := percentile(bootstrapMeans, 2.5);

% ... calculate the 90% confidence interval

ci1 := percentile(bootstrapMeans, (100-90)/2);   % = 5 %
ci2 := percentile(bootstrapMeans, (100+90)/2);   % = 95 %

confidenceIntervall := {ci1,ci2};
