procedure tCDF(a,df);
   if a=0 then 0.5 else 
   if a<0 then 0.5 * ibeta(df/2, 0.5, df/(a^2+df)) else
           1 - 0.5 * ibeta(df/2, 0.5, df/(a^2+df));
          
% ------------------- 0. version  ---------------------

% a rough 'solution', which only give an !integer! approx.
% using gigant steps of size 1 ;) - this is just for fun:

procedure tINV0(alpha,df);
   begin scalar k := 0;
         repeat k := k+1 until tcdf(k,df) >= alpha;
         return k;
   end$

on rounded$
print_precision 5$

tINV0(0.99,3);     % = 5 ~ 4.5407  ~ok~ 
                   %-- (!) .99 NOT allowed


% -------------- 1. version: bisection search -----------

% we only need t-cumulative distribution tCDF as helper
% we show only a simple version,
% given by ChatGPD with minor corrections
 
procedure tCDF(a,df);
   if a=0 then 0.5 else 
   if a<0 then 0.5 * ibeta(df/2, 0.5, df/(a^2+df)) else
           1 - 0.5 * ibeta(df/2, 0.5, df/(a^2+df));

% Inverse Student-t distribution using bisection method

procedure tinv1(p,nu);
   begin scalar lo:=0, hi:=100, eps:= 1.0e-10,
                mid,flo,fmid;
      while hi-lo > eps do
         begin mid  := (lo+hi)/2;
               fmid := tcdf(mid,nu);
               if fmid < p then lo:=mid else hi:=mid;
         end;
      return (lo+hi)/2;
   end;

on rounded$
print_precision 5$

tINV1(0.99,3);      % = 4.5407  ok 
tinv1(0.975, 10);   % should be close to 2.228
tinv1(0.95,1);      % should be ≈ 6.3138
