procedure SUM1(X); 
          for each el in X sum el $

procedure DOT(A,B); 
          for i := 1:length(A) sum part(A,i)*part(B,i)$
          
procedure meanF(X,f); 
          dot(X,f)/Sum1(f)$

procedure momentF(X,f,r);
begin
  scalar n, mf, diff;
  n    := length(X);  
  mf   := meanF(X,f);
  diff := for each el in X   collect (el - mf)^r;
  return dot(f, diff) / sum1(f)
end;

f := {1,4,6,10,7,2};
