load_package excalc;

pform {x,y,t} = 0, {dx,dy,ds} = 1;
x := t;
y := t;

@(t) _| d x;

% Line element
ds := sqrt( (@(t) _| d x)**2 + (@(t) _| d y)**2 );   

% Integral
int( x*ds, t, 0, 1);


end;
