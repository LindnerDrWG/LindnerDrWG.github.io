procedure weibullCDF(x,a,b);
          if (x<0) then 0.0 else  1 - exp(-(x/a)^b) $


% ------------------- Example usage ---------------------

on rounded $
print_precision 5$

weibullCDF(0.5, 0.15, 0.8);
