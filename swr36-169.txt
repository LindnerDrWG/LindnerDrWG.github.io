COMMENT 
/*  cdf of a Pareto distribution with parameters 
-- m='mu'=location and a='alpha'=shape parameter.
-- https://search.r-project.org/CRAN/refmans/EnvStats/html/Pareto.html
-- We adpot this R convention. */ $


procedure paretoCDF(x,m,a);
          if  x<m then  0.0 else 1 - (m/x)^a $


% ------------------- Example usage ---------------------

on rounded $
print_precision 5$

1 - paretoCDF(5, 1, 2.5) ;  % = 0.017889

paretoCDF(4, 1, 1) ;
