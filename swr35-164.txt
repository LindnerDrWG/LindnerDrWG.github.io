procedure chi2INV(p,k);
           << clear x;
           rhs first num_solve( iGamma(k/2, x/2) = p, x=2)
           >>;
           
% ------------------- Example usage ---------------------

on rounded $
print_precision 5$

for i:=1:10 do write {i/10,  chi2INV(i/10,10)} ;
