procedure chi2INV(p,k);
           << clear x;
           rhs first num_solve( iGamma(k/2, x/2) = p, x=2)
           >>;
           
% ------------------- Example usage ---------------------

on rounded $
print_precision 5$

for i:=1:10 do write {i/10,  chi2INV(i/10,10)} ;

% ------------------- Example with chiINV_bisection ------
% Check with chiINV_bisection or chi2INV_Newton.

procedure chi2CDF(x, k);
          if x<=0 or k<=0 then 0.0 
          else iGamma(k*0.5, x*0.5) $

procedure chi2inv_bi(p,nu);
  begin scalar lo,hi,mid,fmid, tol:= 1.0e-8;
    lo := 0.0; hi := nu;
    % Find an upper bracket hi
    while chi2CDF(hi,nu) < p do hi := 2.0*hi;
    % Bisection
    for i := 1:1000 do
    << mid := (lo+hi)/2.0;
       fmid := chi2cdf(mid,nu);
       if abs(fmid-p) < tol then  mid; 
       if fmid < p then lo := mid else hi := mid;
    >>;
    return (lo+hi)/2.0;
  end;
