procedure fINV(p,df1,df2);
      << clear x;
         rhs first num_solve(iBeta( 0.5*df1, 0.5*df2, x/(x+df2/df1)) = p, x=2)
      >>$

% ------------------- Example usage ---------------------

on rounded $
print_precision 5$

X := {0.25, 0.5, 0.75, 0.999};
for each i in X do write {i, finv(i,10,20)};
