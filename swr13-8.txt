procedure SD(lst);
begin
  scalar n, mu, diff_sum;
  n := length(lst);  
  if n = 0 then 0;  
  mu := (for each el in lst sum el)/n;  % mean
  diff_sum := for each el in lst sum (el - mu)^2;
  return sqrt(diff_sum/n)
end;

% ------------------- Example usage ---------------------

on rounded$
print_precision 5$

X := {12,6,7,3,15,10,18,5};

sd(X);
