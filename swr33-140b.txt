%          We show only a simple version, asking e.g. ChatGPT:
COMMENT          Convergence guarantee: 

  The Newton method requires a robust starting value. 
  Since the t-distribution is smooth and bell-shaped, 
  a starting value of 0.0 works perfectly for most standard 
  quantiles 0.1 < p < 0.9.
  The starting guess tan(pi*(p-0.5)) is exact for nu=1 (Cauchy) 
  and a good sanity check: tinv(0.95,1) should return ≈6.3138, 
  converging in a single Newton step.
  Since the cumulative distribution function (CDF) of the 
  t-distribution is strictly monotonically increasing, (fig.!)
  the equation tCDF(x,nu)-p=0 can be solved exactly and stably. $


% -------------- 2. version: NEWTON iteration (ChatGPT) -----------
%  we need both tpdf and tcdf as helper, see fig. left

procedure tPDF(x,f);  
          1/(sqrt(f)*Beta(1/2,f/2))*(1+ x^2/f)^(-0.5*(f+1)) $

procedure tCDF(a,df);
   if a=0 then 0.5 else 
   if a<0 then 0.5 * ibeta(df/2, 0.5, df/(a^2+df)) else
           1 - 0.5 * ibeta(df/2, 0.5, df/(a^2+df));

% Inverse Student-t distribution using Newton iteration
% given by ChatGPD, minor corrections

procedure tinv2(p,nu);
   begin scalar x, xnew, f, fx, n,
      tol:= 1.0e-10, maxit:= 100;
      x := 0; % Initial approximation
              % x := tan(pi*(p-0.5)); would be a  
              % Cauchy-based starting guess, exact when nu=1
      % Newton iteration
      for n := 1:maxit do
        begin
          f  := tcdf(x,nu) - p;  % nominator
          fx := tpdf(x,nu);      % denominator
          xnew := x - f/fx;
          if abs(xnew-x) < tol then return xnew;
          x := xnew
        end;
      return x
   end;

% ----- Example usage

on rounded$
print_precision 5$

tINV2(0.99,3);      % = 4.5407  ok 
tinv2(0.975, 10);   % should be close to 2.228
tinv2(0.95,1);      % should be ≈6.3138



% -------------- 2a. version: NEWTON iteration by fjW -----------

COMMENT by F.J. Wright in private communication:
  The main problem with the Newton-Raphson code in 2. is that it 
  always executes maxit iterations because the return in the inner 
  begin...end only stops the begin...end containing it, not the loop. 
  Also, it is ignoring the return value (and it really only makes
  sense to return a local value).   $

procedure tPDF(x,f);  
          1/(sqrt(f)*Beta(1/2,f/2))*(1+ x^2/f)^(-0.5*(f+1)) $

procedure tCDF(a,df);
   if a=0 then 0.5 else 
   if a<0 then 0.5 * ibeta(df/2, 0.5, df/(a^2+df)) else
           1 - 0.5 * ibeta(df/2, 0.5, df/(a^2+df));

procedure tinv2a(p,nu); 
   begin scalar x, f, fx, dx,
      tol:= 1.0e-8, maxit:= 100;
      x := 0;  
      repeat <<
         f  := tcdf(x,nu) - p;
         fx := tpdf(x,nu);
         dx := f/fx;
         x := x - dx;
         maxit := maxit - 1
      >> until abs dx < tol or maxit = 0;
      return x
   end$

% ----- Example usage

on rounded$
print_precision 5$

tINV2a(0.99,3);      % = 4.5407  ok 
tinv2a(0.975, 10);   % should be close to 2.228
tinv2a(0.95,1);      % should be ≈6.3138
