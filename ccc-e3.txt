operator e;                  % operator e() with properties of 2x2 det()
noncom e;                    % e is declared noncommutative    *  e1  e2
for all j let e(j)**2 = 0;                                 %  e1   0   1
let e(2)*e(1) = -e(1)*e(2);                                %  e2  -1   0

v := (a*e(1)+b*e(2));
w := (c*e(1)+d*e(2));

v * w;
w * v;           % switch sign, i.e. 'alternating'

v := 3*e(1)+1*e(2);
w := 1*e(1)+4*e(2);

v*w;             % changing order change the sign of the result
w*v;

det mat((3, 1),  % compare with build-in determinant
        (1, 4));

end;
