% ------------------- helper functions ---------------------
precision 10;
procedure normCDF(x,m,s); 1/2+erf((x-m)/(s*sqrt(2)))/2 $
% ------------------- helper functions ----------------- END


on rounded $
print_precision 5 $

% parameters of sample */
x_bar:= 105;
mu:= 100;
sigma:= 15;
n:= 50;
alpha:= 0.05; 

% 1. calculate test statistics T = z_value */
z_value:= (x_bar - mu) / (sigma / sqrt(n));

% 2. calculate p-value for two sided test */
%    normCDF(z, 0, 1) gives area up to the z-value

p_value:= 2 * (1 - normCDF(abs(z_value), 0, 1));

% 3. make decision */
if p_value < alpha then
    write("H0 is rejected: the mean value is significant different. p-value = ", p_value)
else
    write("H0 will be retained: no significant difference. p-value =", p_value);
