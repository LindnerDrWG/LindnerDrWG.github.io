procedure moment(X,A,r);
begin
  scalar n, mom;
  n   := length(X);  
  mom := for each el in X  sum (el - A)^r;
  return mom/n
end;

% we may define it as a one-liner:

procedure moment(X,A,r);
         (for each el in X  sum (el - A)^r )/length(X);


% ------------------- Example usage ---------------------

on rounded $
precision 5$

X := {4,36,45,50,75}$

moment(X,0,1);
moment(X,0,2);
moment(X,4,3);
