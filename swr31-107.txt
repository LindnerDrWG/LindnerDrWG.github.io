procedure normCDF(x,m,s); 
          1/2+erf((x-m)/(s*sqrt(2)))/2 $


% =======================================================
%  Example usage
% =======================================================

on rounded$
print_precision 5$

for i:=-2:2 do write {i, normCDF(i, 2,1)};
