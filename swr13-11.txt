procedure XAPPLY(f,lst); lisp(f.cdr(lst))$
procedure SUM1(lst);     xapply(+, lst)$
procedure MEAN(lst);     xapply(+, lst)/length(lst)$

procedure SD(lst);
begin
  scalar n, mu, diff_sum;
  n        := length(lst);  
  if n = 0 then 0;  
  mu       := for i := 1:n sum (part(lst, i)/n);
  diff_sum := for i := 1:n sum (part(lst, i) - mu)**2;
  return sqrt(diff_sum/n)
end;
