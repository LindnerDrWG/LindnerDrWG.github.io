% ------------------- helper functions ---------------------
COMMENT
   To calculate the tDIFF score 'T' we need only function
   tCDF()  $

procedure tPDF(x,f);  
          1/(sqrt(f)*Beta(1/2,f/2))*(1+ x^2/f)^(-0.5*(f+1)) $
procedure tCDF(a,df);
   if a=0 then 0.5 else 
   if a<0 then 0.5 * ibeta(df/2, 0.5, df/(a^2 + df)) else
           1 - 0.5 * ibeta(df/2, 0.5, df/(a^2 + df));
procedure MEAN(X);  (for each el in X sum el)/length(X);
procedure SUM_(X);  (for each el in X sum el);
define    DIM = length$


%---------------------------------------------------
procedure  tdiff(X,Y,alpha);
%---------------------------------------------------
begin scalar T_,df,D,d_,n, sum1, sum2, pvalue;
   n  := dim(X);
   df := n-1;
   D  := for k:=1:n collect part(X,k)-part(Y,k); 
   sum1 := Sum_(D);
   sum2 := for k:=1:n sum part(D,k)*part(D,k);  % = D.D
   d_  := sqrt((sum2-sum1^2/n)/(n*(n-1.)));
   T_   := sum1/(n*d_);
   pValue := min( 2*tCDF(T_,df), 2*(1-tCDF(T_,df)) );
   if (pValue < 0.975) then write("..... accept H0")
                        else write("..... reject H0");
   write "T = ",T_;
   write "pValue = ",pvalue;
   mat((T__,p_alue),(T_, pvalue))    %-- no output                                  
END;
