% ---------------------- helper functions ------------------
procedure sum_(lst);   for each el in lst sum el $ 
procedure mean(lst);   sum_(lst)/length(lst) $ 
procedure listmake(m); for i:=1:m collect 0$
define    dim    = length $


%### ---------------------------------------------------------
%###     ALGORITHM: Jackknife method '1-deleted'
%### ---------------------------------------------------------

procedure jack(X, alpha);
begin scalar n,mX,mM,jac,D,su,sigma,SE,M,JCK;
    n := dim(X);
    M := listmake(n);
    mX := mean(X);
    for i:=1:n do
        begin
          h := part(X,i); 
          X := (part(X,i):=0);           
          M := (part(M,i):=n/(n-1)*Mean(X));   
          X := (part(X,i):=h)
        end;
    mM  := mean(M);            
    jac := n*mX - (n-1)*mM ;  
    JCK := for i:=1:n collect   n*mX - (n-1)*part(M,i);    
	D   := for i:=1:n collect   jac - part(JCK,i);
	su  := for i:=1:n Sum       part(D,i)^2; 
	sigma := sqrt(su/(dim(X)-1));  
    SE  := sigma/sqrt(n);   
    CI  := {Mean(JCK) - SE, Mean(JCK) + SE};
    % alternative: CI := tINV(alpha/2,n-1)*SE
    write("The nonparametric -leave-one-out- jackknife algorithm");
    write "SE = ", se;
    write "CI = ", CI
    % load_package matrix;
    % mat((SE_, CI_l, CI_h), (SE, mean(JCK)-SE, mean(JCK)+ SE))
END;
