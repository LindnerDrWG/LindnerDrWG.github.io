procedure chi2PDF(x, k);
          1/(2^(k/2)*gamma(k/2))*x^(k/2-1)*exp(-x/2.) $


procedure chi2CDF(x, k);
          if x<=0 or k<=0 then 0.0 
          else iGamma(k*0.5, x*0.5) $

% Inverse chi-squared CDF using Newton iteration
procedure chisqinv(p,nu,x0,tol,maxit);
   begin scalar x,fx,dfx,xnew,k;
      x := x0;
      % for k := 1:maxit do
      repeat
      <<
         fx   := chi2cdf(x,nu) - p;
         dfx  := chi2pdf(x,nu);
         x    := x - fx/dfx;
         maxit := maxit - 1
         >> until abs x < tol or maxit = 0;
         %if abs(xnew-x) < tol then return xnew;
         %if xnew <= 0 then  xnew := x/2;
         %x := xnew;
      end;
      return x
   end;

         f  := chi2CDF(x,nu) - p;
