on rounded; precision 10;

procedure normINV(q,m,s);
     if q=0 then  infinity else
     if q=1 then -infinity else
     if 0<q and q<1 then if q = 0.5 then 0 else
     begin scalar y := 2*q-1, xx;
        clear x;
        xx := num_solve(erf x = y, x = y);
        xx := rhs first xx;
        return m + sqrt(2) * s * xx;
    end$

procedure stdnormINV(q); normINV(q,0,1) $

% ---------- inverse t distribution ------------------------

procedure tINV1(x,f);
%-- after Preizer & Pratt
%-- t reserved symbol for t(rue); therefore sysmbol tt
begin scalar q,c, tt;
      if f=1 then return tan(PI*(x-0.5));
      if f=2 then return <<tt:=2.*x-1.; sqrt(2.)*tt/sqrt(1.-tt^2)>>;
      if x = 0 or x = 1 then tt := infinity
      else <<                      %-- 0<x<1
         q := stdnormINV(x);
         c := f - 2/3 + 0.1/f;
         c := (f-5/6)/c^2;
         tt := sqrt(f*exp(c*q^2)-f);
      >>;
      return if x>0.5 then tt else -tt;
end;

% -------------------  Example usage     -------------------

tINV1(0.99,3);   %% -- 4.54 vs 4.61

%% An abbreviated input like .95 is not allowed, write 0.95:
tinv1(0.95,50);  %% 1.6759, ok bis x.xxxxx
                 %% MatLab>  ans = 1.6759
                 %% R>       qt(.95,50) [1] 1.675905

for k:=0:10 do write "tINV1(",k/10,",5) := ", tINV1(k/10,5);
