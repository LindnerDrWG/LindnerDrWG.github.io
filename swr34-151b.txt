procedure fINV(p,df1,df2);    
   % solve fcdf(x) = p for x, i.e. find the p-quantile
   <<
     clear x;
     rhs first num_solve( iBeta( 0.5*df1, 0.5*df2, x/(x+df2/df1)) = p, x=2)
   >>$


% ------------------- Example usage ---------------------

on rounded $
print_precision 5$

finv(0.95,5,10);    % =3.3258
