COMMENT ---------------------------------------------
   We first show an interactive session for the
   successive construction of function MODE.
   You may copy & paste the following commands line-by-line
   into the Input Editor of WebReduce.
----------------------------------------------------- ;

% we want to have the MODE of list X
X := {1,3,6,6,6,6,7,7,12,12,17};
 
% this is the fequency list of X 
L:= {{1,1},{3,1},{6,4},{7,2},{12,2},{17,1}};

COMMENT The idea is to get the maximal value (6,4)
        of all pairs and choose its first entry 6 as MODE
        
% to start, we pick the FIRST element of L by part(L,1); or
first(L);

% this element is our first candidate for the MODE
maxx:=first(L);
part(maxx,2);     
 
% we now seek for a better pair 
for i:=1:6 do if   part(part(L,i),2) > part(maxx,2) 
              then maxx:=part(L,i);
 
% now we have our maximal pair whose 1st entry is the MODE 
part(maxx,1);
 
COMMENT ---------------------------------------------
   We abstract this process/procedure to a function MODE:
----------------------------------------------------- ;

procedure MODE(X);
begin scalar maxx, L;
      load_package assist;
      L := frequency(X);
      maxx:=first(L);
      for i:=1:length(L) do 
          if part(part(L,i),2) > part(maxx,2) 
          then maxx:=part(L,i);
      return part(maxx,1)
end;
