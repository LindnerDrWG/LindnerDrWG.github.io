% --------------------- helper functions ----------------
procedure SUM_(X);  for each el in X sum el$ 
procedure mean(X);  Sum1(X)/length(X)$
procedure listmake(m); for i:=1:m collect 0$
procedure sp(X,Y);  % scalar product alias inner product
   for i := 1:length(X) sum part(X,i) * part(Y, i)$
procedure quicksort(l);  % - by Gemini AI
   if l = {} then {}
   else begin
      scalar pivot, rest_l, left, right, x;
      pivot := first(l);  rest_l := rest(l);
      left  := {};         right := {};
      for each x in rest_l do
         if x <= pivot then left  := x . left
                       else right := x . right;
      return append(quicksort(left), pivot . quicksort(right));
   end;
procedure putNTH(l,k,v);
 begin scalar a,b; integer i;
    a:=l; b:={};
    for i:=1 step 1 until k-1 do
         <<b:=append(b,{first a}); a:=rest a>>;
    b:=append(b,{v}); a:=rest a;
    return append(b,a)
 end;
procedure crank_aux(v, sL, nL);           
% v : element of list L, sL: sorted L; nL: length L    
% crank.red - Numeric Recipes CRANK, p.510 for REDUCE
% Lisp'ish VERSION: W must be sorted increasingly
% returns S, and W is modified in place with averaged ranks  
   begin scalar positions:={};
      for k:=1:nl do
          if part(sl,k) = v 
             then positions := append(positions, {k});
      return sum_(positions) / length(positions)
    end;
procedure  crank(w);     % w should be sorted
begin scalar s, n;
    s := quicksort(w);   % or sortnumlist(w) of package assist
    n := length(w);
    return for i:=1:n collect crank_aux( part(w,i), s, n) 
END ;
procedure correspond(X,soX);  
begin scalar n:=length(X);
   M  := for i:=1:n collect 0;
   rX := cRank(soX);
   for j:=1:n do for i:=1:n do 
       if    part(soX,i)=part(X,j) 
       then  M := putnth(M, j, part(rX,i));
   return M;
END;

COMMENT
-- We study the example for the Wilcoxon signed rank test 
-- for paired observations from https://nbisweden.github.io/..
-- We cite the original comments from ibid. in the script.

-- Going back to our sleep study in 4.10, now we are ready 
-- to examine whether there is enough evidence to reject 
-- a null hypothesis of median of the differences between
-- the paired observations is equal to 0. 
-- Being able to reject a null hypothesis would mean 
-- that the drug is having an effect.

-- SOLUTION: step-by-step.
-- Formulate the null hypothesis Ho.
-- Ho: the median difference in the population equals to zero.
-- SOLUTION:      $

COMMENT ADVICE: submit each of the following command lines 
        (separated by blank lines) one after the other to

on rounded $
print_precision 5$

X := {6.1, 6.0, 8.2, 7.6, 6.5, 5.4, 6.9, 6.7, 7.4, 5.8};
Y := {5.2, 7.9, 3.9, 4.7, 5.3, 7.4, 4.2, 6.1, 3.8, 7.3};

%-- length of sample
n  := length(X) ;             




COMMENT  .................................................
-- Now do 
-- 1. start by calculating the differences between hours  
--    of sleep for each study participant. 
-- 2. exclude any difference that is equal to 0.
-- 3. rank the rest of values in ascending order.
-- 4. ignore the signs (i.e. use absolute values)
-- 5. calculate the test statistics T = min(T+,T-)
%........................................................ $



%-- differences X-Y between hours of sleep 
D := for i:=1:n collect part(X,i)-part(Y,i);   

%-- ignoring the sign of D
aD := abs(D) ;      

%-- sort the absolute differences
sD := sort(ad);     


%-- the helper function 'correspond' associates X ~ sX, 
%-- i.e.  X is reordered w.r.t the sorted sX 
%-- cdDis the ranking of aD, implicit done in correspond

cD := correspond(aD,sD) ;        

%-- id is used for summary table, it enumerates the data
id := for i:=1:n collect i ;

%-- Summary in rows (id, X, Y, d, cd ):
{id, X, Y, D, cD };


COMMENT ..................................................
-- Now calculate the test statistics W:
-- sum up the ranks of the negative differences 
-- and of the positive differences and denote these sums 
-- by T− and respectively T+:
%........................................................ $


Tp := for i:=1:n collect if part(D,i)>0 then 1 else 0;   

Tm := for i:=1:n collect if part(D,i)<0 then 1 else 0;  

T0 := for i:=1:n collect if part(D,i)=0 then 1 else 0; 
% to is forbitten because of .. to .. ;)

%-- calculate the test statistics 
%--   summing via inner product  SP=DOT

Wp := dot(Tp, cd);     

Wm := dot(Tm, cd);

COMMENT
-- If there were no differences in effectiveness between 
-- the sleeping drug and the placebo, then the sums  T− 
-- and T+ would be similar. 
-- Test statistics: denote the smaller sum by T and 
--                  interpret the p-value        $


T_ := min(Wp,Wm);


Comment   DECISION
-- The hypothesis is that T is equal to the sum of the 
-- ranks divided by 2, so that the smaller T the more 
-- evidence is there against the null hypothesis.   $

Wcrit05 := {_,_,_,_,_,_, 2,4,6,8,11,14,17,21,
          25,30,35,40,46,52,59,66,73,81,89}$

wc := part(Wcrit05, 10);  %-- use 0.025 for one-sided test

if(T_ > wc) then write("accept H0") 
            else write("reject H0");

COMMENT
-- In our example, T=15 and the sample size is n=10, 
-- where n is the number of non-zero differences (we had none). 
-- According to the W-table, the 5% percentage point is at wc = 8. 
-- Since T = 15 > 8, we conclude that our p-value is > 0.05
-- and we do not have enough evidence to reject the null hypothesis, 
-- i.e.: there is no evidence of the sleeping drug working.
$
