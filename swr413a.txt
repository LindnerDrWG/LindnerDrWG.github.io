%|                  PEARSON's Chisq test

% ------------------- helper functions ---------------------
procedure chi2CDF(x, k);
          if x<=0 or k<=0 then 0.0 
          else iGamma(k*0.5, x*0.5) $
procedure chi2inv(p,nu);  % use only chi2CDF
  begin scalar lo,hi,mid,fmid, tol:= 1.0e-10;
    lo := 0.0; hi := nu;
    while chi2CDF(hi,nu) < p do hi := 2.0*hi;
    for i := 1:1000 do
    << mid := (lo+hi)/2.0;   % bisection
       fmid := chi2cdf(mid,nu);
       if abs(fmid-p) < tol then  mid; 
       if fmid < p then lo := mid else hi := mid;
    >>;   return (lo+hi)/2.0;
  end;


% ###########################################
%       EXAMPLE: vitamin C prophylaxis
% ###########################################

on rounded $
print_precision 5$

a:=17; b:=122; c:=31; d:=109;   %-- the data of the table 
n  := a+b+c+d;
z  := abs(a*d-b*c) - 0.5*n;  %-- YATES correction ..-0.5*n

chi := z*z*n/((a+b)*(c+d)*(a+c)*(b+d));
p   :=  1 - chi2CDF(chi,1);    %--  p-value nach R
phi := sqrt(chi/n);            %-- contingency coefficient

%-- test statistics 

mat((chi_,p_,phi_),(chi,p,phi));

write "chi = ", chi;
write "p = ", p;
write "phi = ", phi;

%-- pval = 1 - chi2CDF(chi, 1)    
p    := 1 - chi2INV(1-0.05/2, 1);

COMMENT result
