procedure SUM1(X); 
          for each el in X sum el$
          
procedure mean(X); 
          Sum1(X)/length(X)$

procedure cov(X,Y) ;
begin
  scalar n, mX, mY, co;
  n   := length(X);  
  mX  := mean(X);
  mY  := mean(Y); 
  co := for i := 1:n sum (part(X, i) - mX)*(part(Y, i) - mY);
  return co/n
end;

procedure covR(X,Y) ;   % method 'pearson
begin
  scalar n, mX, mY, co;
  n   := length(X);  
  mX  := mean(X);
  mY  := mean(Y); 
  co := for i := 1:n sum (part(X, i) - mX)*(part(Y, i) - mY);
  return co/(n-1)
end;

cov(X,Y);
