%----------------------------------------------------------
%          EXAMPLE: heigths of fathers vs sons
%----------------------------------------------------------
% PROBLEM: The heights of a sample of 12 fathers and their oldest 
%          sons are measured giving the following data:
%          Far : 65,63,67,64,68,62,70,66,68,67,69,71
%          Son : 68,66,68,65,69,66,68,65,71,67,68,70
%          cf. Spiegel p. 434 P 17.24
% TASK:    Calculate the Spearman rank correlation rho ρ.

%-- a. SOLUTION manually step-by-step by REDUCE-'hand'

on rounded $
print_precision 5$

Far := {65, 63, 67, 64, 68, 62, 70, 66, 68, 67, 69, 71};
Son := {68, 66, 68, 65, 69, 66, 68, 65, 71, 67, 68, 70};

%-- we may sort both data sets by eye, because they are small :)

sFar := {62, 63, 64, 65, 66, 67, 67, 68, 68, 69, 70, 71};
sSon := {65, 65, 66, 66, 67, 68, 68, 68, 68, 69, 70, 71};

COMMENT
  -- Because sFar[4]=65, 65 has rank=4, 
  --      we  set 4 in place Far[1]=65, etc.
  --            1.
  -- Far =     (65, 63,  67, 64, 68, 62, 70, 66, 68,  67,  69, 71)
  --             |   |  ...
  -- rankFar = ( 4,  2, 6.5,  3, 8.5, 1, 11,  5, 8.5, 6.5, 10, 12)
  -- we rank equal data using mean values if 'ties' are seen!
$

rFar := { 4,    2, 6.5,   3, 8.5,   1,  11,   5, 8.5, 6.5,  10, 12};
rSon := {7.5, 3.5, 7.5, 1.5,  10, 3.5, 7.5, 1.5,  12,   5, 7.5, 11};

%-- we now calculate the differences of corresponding ranks 
n := dim(Far);
D := for i:= 1:n collect part(rFar,i) - part(rSon,i);

%-- and their squared values D^2 to get positive signs
D2 := for i:= 1:n collect part(D,i)^2;

%-- we have row 1: D, row2 : D^2:
{D, D2};

procedure SUM_(X);  for each el in X sum el$ 

sum_(D2);      % = 72.5


%-- According to [1], Spearman's correlation rho is defined as 
%--    r = 1 - 6*sum((ranks(x)-ranks(y)).^2)/(N*(N*N-1))

rho_Spearman := 1 - 6*sum_(D2)/n/(n^2-1);

procedure quicksort(l);  % - by Gemini AI

   else begin
      scalar pivot, rest_l, left, right, x;
      pivot := first(l);  rest_l := rest(l);
      left  := {};         right := {};
      for each x in rest_l do
         if x <= pivot then left  := x . left
                       else right := x . right;
      return append(quicksort(left), pivot . quicksort(right));
   end;
   
% crank.red - Numeric Recipes CRANK, p.510 for REDUCE
% Lisp'ish VERSION: W must be sorted increasingly
% returns S, and W is modified in place with averaged ranks        

procedure crank_aux(v, sL, nL);   
   % v : element of list L, sL: sorted L; nL: length L      
   begin scalar positions:={};
      for k:=1:nl do
          if part(sl,k) = v 
             then positions := append(positions, {k});
      return sum_(positions) / length(positions)
    end;
procedure  crank(w);     
begin scalar s, n;
    s := quicksort(w);   % or sortnumlist(w) of package assist
    n := length(w);
    return for i:=1:n collect crank_aux( part(w,i), s, n) 
END ;

% -- We need Helper function putNTH.

procedure putNTH(l,k,v);
 %     putNTH sets k-th element of list L to value v
 %     part(L,k) gives only _reading_ access
 begin scalar a,b; integer i;
    a:=l; b:={};
    for i:=1 step 1 until k-1 do
         <<b:=append(b,{first a}); a:=rest a>>;
    b:=append(b,{v}); a:=rest a;
    return append(b,a)
 end;

write M;

for j:=1:dim(Far) do 
for I:=1:dim(Far) do if part(sFar,i)=part(Far,j)

COMMENT
 -- NOW we abstract this process into a function 'correspond'
 -- Because sX is a turtle operator, we use soX for the
 -- sorted list X   $

procedure correspond(X,soX);  
begin scalar n:=length(X);
   M  := for i:=1:n collect 0;
   rX := cRank(soX);
   for j:=1:n do for i:=1:n do 
       if    part(soX,i)=part(X,j) 
       then  M := putnth(M, j, part(rX,i));
   return M;
END;

cS := correspond(Son, sSon);
