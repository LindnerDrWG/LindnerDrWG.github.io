procedure chi2CDF(x, k);
          if x<=0 or k<=0 then 0.0 
          else iGamma(k*0.5, x*0.5) $

% ------------------- Example usage ---------------------

on rounded $
print_precision 5$

%----- a.

for x:=0:8 do write {x, chi2CDF(x,4)};
