procedure fCDF(x,df1,df2);
          iBeta( 0.5*df1, 0.5*df2, x/(x+df2/df1))$

% ------------------- Example usage ---------------------

on rounded $
print_precision 5$

for x:=0:8 do write {x/2, fCDF(x/2,5,2)};
