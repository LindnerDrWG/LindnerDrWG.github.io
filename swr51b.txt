%|         corrCoeff

procedure MEAN(lst); 
          (for each el in lst sum el )/length(lst)$
procedure sumSQ(lst);  for each el in lst sum el^2 $
procedure SP(X,Y);  % scalar product alias inner product
          for i := 1:length(X) sum part(X,i) * part(Y, i)$
                             % zip the sum of products 
define    DOT = SP$

%--------------------------------------------------------
%      DEFINITION (CORRELATION Coefficient rho alias r)
%--------------------------------------------------------

procedure corrcoef(X,Y);
begin scalar nom:=0,denom:=1, Xm,Ym;
   xbar := Mean(X);
   ybar := Mean(Y);
   Xm   := for each el in X collect (el - xbar); 
   Ym   := for each el in Y collect (el - ybar); 
   num  := dot( Xm, Ym);  
   den  := sqrt( sumsq(Xm) * sumsq(Ym) ) ; 
   return num/den
END;

procedure r(X,Y); corrcoef(X,Y)$
%--------------------------------------------------------


%      EXAMPLE of Pearson's product-moment formula 

X := {60,65,72,70,74,63,66,68,67,69,70,64}$
Y := {120,140,180,184,190,160,155,158,170,190,210,150}$

on rounded $
print_precision 5$

corrcoef(X,Y);
r(X,Y);
