procedure SUM1(lst); 
          for each el in lst sum el $
             
procedure MEAN(lst); 
          sum1(lst)/length(lst);

procedure SD(lst);
begin
  scalar n, mu, diff_sum;
  n := length(lst);  
  if n = 0 then 0;  
  mu := mean(lst);
  diff_sum := for each el in lst sum (el - mu)^2;
  return sqrt(diff_sum/n)
end;

 
procedure zscore(lst);
begin
  scalar n, mu, Xmu, zscore1;
  n       := length(lst);  
  if n = 0 then 0;  
  mu      := mean(lst);
  sdL     := sd(lst);
  Xmu     := for each el in lst  collect (el - mu);
  zscore1 := for each el in Xmu  collect el/sdL;
  return zscore1
end;
