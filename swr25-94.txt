% WARNING: do not use the reserved symbol lambda

procedure poissonPDF(k, lam);
          exp(-lam) * lam^k / factorial(k);


% -------------------- Example usage ---------------------

on rounded$
print_precision 5$

for k:=0 step 1 until 10 do write{k, poissonPDF(k,2.5)} ;
