% ------------------- helper functions ---------------------
%   To calculate the 2-Z-TEST we need the functions
%   stdnormCDF() and stdnormINV() as helper.    
precision 10;
procedure normCDF(x,m,s); 1/2+erf((x-m)/(s*sqrt(2)))/2 $
procedure stdnormCDF(x);  normCDF(x, 0,1);   
procedure normINV(q,m,s);
   if q=0 then  infinity else if q=1 then -infinity else     
   if 0<q and q<1 then
      begin scalar y := 2*q-1, xx;
        if y = 0 then return 0;
        clear x;
        xx := num_solve(erf x = y, x = y); % a list of equations
        xx := rhs first xx;
        return m + sqrt(2) * s * xx
       end$
procedure stdnormINV(q); normINV(q,0,1)$
procedure MEAN(X);  (for each el in X sum el)/length(X);
define dim = length$
procedure VAR(lst);
   begin
     scalar n, mu, ds;
     n := length(lst);  
     mu := (for each el in lst sum el) / n;
     ds := for each el in lst sum (el - mu)^2;
     return ds/n
   end;
procedure SD(lst); sqrt(var(lst));

% ------------------- helper functions ----------------- END

% ------------------- function ZTEST ---------------------
procedure ztest2(X,Y, muX, muY, sigmaX, sigmaY, alpha, altHyp) ;
%    output: (Z-value, quantile, ConfidenceInterval, p-value) 
begin scalar nX,nY,Z,SEdiff;
   nX  := dim(X);
   nY  := dim(Y);
   Z := ((mean(X)-mean(Y))-(muX-muY))/sqrt(sigmaX^2/nX+sigmaY^2/nY);
   SEdiff := sqrt(sigmaX^2/nX + sigmaY^2/nY);  %-- std.error 
      
   if not (althyp = 1 or althyp = -1 or althyp = 0) 
      then rederr("ztest: altHyp must be 0,1 or -1.");
      
   if altHyp = 0 then                     %-- two-sided test 
         <<
            Zcrit := stdnormINV(1-alpha/2);  %-- alpha-quantile 
            CI := {(mean(X)-mean(Y)) - Zcrit*SEdiff, 
                   (mean(X)-mean(Y)) + Zcrit*SEdiff };
            p  := min( 2 * stdnormCDF(Z), 2 * (1-stdnormCDF(Z)) )
          >>;
   if altHyp = 1 then                %-- one-sided test greater 
         <<
            write "Your task"
            >>;
   if altHyp = -1 then               %-- one sided smaller 
         <<
            write "Your task"
          >>;

write "Z          = ", Z;
write "Zscore     = ", Zcrit;
write "CI_left  = ", first CI;
write "CI_right = ", second CI; 
write "          p = ", p;
