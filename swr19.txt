%-- We use the 'nearest-rank method' for calculating the quantile.

procedure quantile(X,p);              % improved by fjW
begin scalar Xs, i, dim;
      if p >= 100 then return max(X);
      if p <= 0 then return min(X);
      dim := length(X);
      Xs  := sort(X, <);
      i   := 1 + round((dim-1)*p/100);
      return part(Xs, i)
end;

% ------------------- Example usage ---------------------


X:={12, 3, 4, 56, 78, 18, 46, 78, 100};

on rounded $
print_precision 5$

quantile(X, 25);
quantile(X, 50);
quantile(X, 75); 
quantile(X, 100);
  
end;
