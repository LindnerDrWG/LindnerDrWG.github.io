% Remember:
%
%      fCDF(x,df1,df2) =
%          iBeta( 0.5*df1, 0.5*df2, x/(x+df2/df1))    (1)
%

% we follow the calculation of fINV(..)  step-by-step
% using the defining term (1)

% ------------------- Example usage ---------------------
% with direct use of iBeta term

on rounded $
print_precision 5$

load numeric;
p     := 0.95; 
df1   :=  8; 
df2   := 10;
f_eqn := iBeta( 0.5*df1, 0.5*df2, x/(x+df2/df1)) = p;
xx    := num_solve(f_eqn, x = 2.0);
xx    := rhs first xx;
write "Inverse F-value: ", xx;
%% [1] 3.071658
