%|         WILCOXON one sample signed rank test 

% ----------------------- helper functions ----------------
procedure SUM_(X);  for each el in X sum el$
procedure SP(X,Y);  % scalar product alias inner product
   for i := 1:length(X) sum part(X,i) * part(Y, i)$

procedure sortinc(l); sort(l, <);

procedure crank_aux(v, sL, nL);
   begin scalar positions:={};
      for k:=1:nl do
          if part(sl,k) = v
             then positions := k . positions;
      return sum_(positions) / length(positions)
   end;

procedure crank(w);
   begin scalar s, n;
       s := sortinc(w);
       n := length(w);
       return for each el in w collect crank_aux(el, s, n)
   end;

procedure correspond(X,soX);
   begin scalar n:=length(X);
       M  := for i:=1:n collect 0;
       rX := cRank(soX);
       for j:=1:n do for i:=1:n do
            if    part(soX,i)=part(X,j)
            then  M := (part(M,j):=part(rX,i));
      return M;
   end;

define    DIM  = length $
define    DOT  = sp $


%|         WILCOXON one sample signed rank test with REDUCE
%###########################################################
procedure  signrank(X,med);
%###########################################################
BEGIN scalar n:=length(X),D,sD,aD,cD,Z,ZR, T_;
    D  := for i:=1:n collect part(X,i) - med;  
    aD := map(abs, D); 
    sD := sortinc(aD);
    cD := correspond(aD,sD);
    Z  := for i:=1:n collect if part(D,i)>0 then 1 else 0;
    ZR := for i:=1:n collect part(Z,i)*part(cD,i);
    W  := dot(Z,cd);
    T_  := (W - n*(n+1)/4)/ sqrt(n*(n+1)*(2*n+1)/24);
     write " (W,T)=", {W,T_}
END;
%####################################################### END
