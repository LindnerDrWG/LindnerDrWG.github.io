procedure SUM1(X); 
          for each el in X sum el;

%   We use the well-known 'inner product' alias the 'dot'-product 
%   of Linear Algebra to solve the problem using the data structure list
 
procedure DOT(A,B); 
          for i := 1:length(A) sum part(A,i)*part(B,i)$
          % part(A,i) means 'i-th element of list A

% ------------------- Example usage ---------------------

f :=  {5,18,42,27,8} $
Sum1(f);

X := {61,64,67,70,73}$

on rounded;
print_precision 5$

dot(X,f)/Sum1(f);

procedure meanf(X,f); 
          dot(X,f)/Sum1(f)$
          
meanf(X,f);
