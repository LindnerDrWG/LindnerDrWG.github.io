procedure SUM1(X); 
          for each el in X sum el$
          
procedure mean(X); 
          Sum1(X)/length(X)$

procedure cov(X,Y) ;
begin
  scalar n, mX, mY, co;
  n   := length(X);  
  mX  := mean(X);
  mY  := mean(Y); 
  co := for i := 1:n sum (part(X, i) - mX)*(part(Y, i) - mY);
  return co/n
end;

procedure covR(X,Y) ;   % method 'pearson
begin
  scalar n, mX, mY, co;
  n   := length(X);  
  mX  := mean(X);
  mY  := mean(Y); 
  co := for i := 1:n sum (part(X, i) - mX)*(part(Y, i) - mY);
  return co/(n-1)
end;

% ------------------- Example usage ---------------------

X := {1,3,5,10};
Y := {2,4,6,20}$

cov(X,Y);    %/* = 23 */

covR(X,Y) ;  %/* = 30.66 */
