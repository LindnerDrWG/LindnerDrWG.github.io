% In this script we use operator part()   
% just to learn to be flexible ;)
     
procedure SUM1(X); 
          for i := 1:length(X) sum part(X, i)$
          
procedure mean(X); 
          Sum1(X)/length(X)$

procedure sd(X);
begin
  scalar n, mx, sx;
  n   := length(X);  
  mx  := mean(X);
  sx := for i := 1:n sum (part(X, i) - mx)^2;
  return sqrt(sx)/(n-1)
end;

procedure moment(X,r);
begin
  scalar n, mx, sx;
  n   := length(X);  
  mx  := mean(X);
  sx := for i := 1:n sum (part(X, i) - mx)^r;
  return sx/n
end;

procedure kurtosis(X);
          moment(X,4)/moment(X,2)^2;   %--- R: with package moments
    
procedure kurtosisM(X);
          moment(X,4)/sd(X)^4;

procedure kurtosisR3(X);
          moment(X,4)/moment(X,2)^2 *(1 - 1/length(X))^2 - 3;
