procedure tCDF(a,df);
   if a=0 then 0.5 else 
   if a<0 then 0.5 * ibeta(df/2, 0.5, df/(a^2 + df)) else
           1 - 0.5 * ibeta(df/2, 0.5, df/(a^2 + df));
          

% ------------------- Example usage ---------------------

on rounded$
print_precision 5$

for i:=-4:4 do write {i, tCDF(i, 5) };

tcdf(3,5) ; %% 0.987
