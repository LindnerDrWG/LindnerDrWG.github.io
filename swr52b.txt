% ----------------------- helper functions ----------------

procedure SORTINC(l);  sort(l, <);
procedure SUM_(X);     for each el in X sum el$ 
procedure LISTMAKE(m); for i:=1:m collect 0$

procedure crank_aux(v, sL, nL);
   begin scalar positions:={};
      for k:=1:nl do
          if part(sl,k) = v
             then positions := k . positions;
      return sum_(positions) / length(positions)
   end;

procedure crank(w);
   begin scalar s, n;
       s := sortinc(w);
       n := length(w);
       return for each el in w collect crank_aux(el, s, n)
   end;

procedure correspond(X,soX);
   begin scalar n:=length(X);
       M  := for i:=1:n collect 0;
       rX := cRank(soX);
       for j:=1:n do for i:=1:n do
           if    part(soX,i)=part(X,j)
           then  M := (part(M,j):=part(rX,i));
      return M;
   end;

% --------------------------------------------------------------
% FUNCTION spearman1 : general function SPEARMAN's rho_S
% --------------------------------------------------------------
procedure  spearman1(X,Y);
   BEGIN   scalar n:=length(X), sX_,sY,cX,cY,D,rho; 
      sX_:= sortinc(X);
      sY := sortinc(Y);
      cX := correspond(X, sX_);
      cY := correspond(Y, sY);
      D  := for i:= 1:n collect part(cX,i) - part(cY,i);
      D2 := for i:= 1:n collect part(D,i)^2;
      rho := 1 - 6*sum_(D2)/n/(n^2-1);
      return rho
   END;
