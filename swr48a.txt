% ------------------- helper functions ---------------------
% -- These problems are dominated by the binomial distribution.
% -- To calculate the sign-TEST score we need the function
%   -- binoPDF() as helper:

precision 10 $
procedure binoPDF(x,n,p);
          binomial(n,x)*p^x*(1-p)^(n-x) $ % x=0,1,2,...n
define    DIM = length$

%#################################################
%     EXAMPLE:  FIRST MARRIAGE 
%#################################################

on rounded $
print_precision 5$

X := {26.0, 25.6, 25.5, 25.6, 25.3, 25.6, 25.7, 
      25.9, 26.0, 26.1, 26.3, 26.6, 26.9, 27.0}$

n  := dim(X) ;       %-- length of samples */
md := 26$            %-- hypothetical median

D := for each el in X collect el-26 ;  % D = X-26

%-- Calculate total number of trials with scores equal to md 26
n26 := 0;
for each el in X do if (el=26.0) then n26 := n26 +1;
write "n26 = ", n26; 
nt:= dim(X) - n26;

%-- Calculate number of successes of scores greater than md=26 
s := 0;
for i:=1:n do if (part(X,i)>26.0) then s := s +1;
s; 

%-- We have:  N+ = 5 = s, N- = 7, N°26 = 2, NTotal = 12 = nt
%-- Now we can calculate the p-value in the case "both-sided":

% -- in R: 
%> dbinom(0, size=12, prob=0.5) + 
%+ dbinom(1, size=12, prob=0.5) + 
%+ dbinom(2, size=12, prob=0.5) + 
%+ dbinom(3, size=12, prob=0.5) + 
%+ dbinom(4, size=12, prob=0.5) +
%+ dbinom(5, size=12, prob=0.5) 
%[1] 0.387207

% binoPDF(2, 12, 0.5) ;
%      0.016113  
%> dbinom(2, size=12, prob=0.5)
%. [1] 0.01611328


p1 := for i:=0:s sum binoPDF(i, nt, 0.5) ; %%%%%%%%%%%%%%%%%%%%%%%%%% ??? 

p2 := 1 - (for i:=0:(s-1) sum binoPDF(i, nt, 0.5) );


p := min(2 * p1, 2 * p2);    %-- Q: 3rd case, (.,.,1) with min.extended


COMMENT -- RESULT:
-- We count 5 differences with a positive sign.
-- The test statistics T is therefore s = 5.
-- The probality p for s = 5 is 0.7744 > 0.05.

-- INTERPRETATION:
-- The null hypothesis, Ho: x_0.5=median=26, can be 
-- rejected with a 21.2 % probability of error. 
$
