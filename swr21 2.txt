procedure binoPDF(x,n,p);
          binomial(n,x)*p^x*(1-p)^(n-x) $
                           

% ------------------- Example usage ---------------------

on rounded $
print_precision 5 $

binoPDF(4, 6, 0.3);      % 0.059535
