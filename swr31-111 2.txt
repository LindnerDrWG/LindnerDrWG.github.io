procedure normCDF(x,m,s); 
          1/2+erf((x-m)/(s*sqrt(2)))/2 $

procedure stdnormCDF(x); normCDF(x, 0,1);    % standard normal CDF 
procedure Phi(x);        stdnormCDF(x);      % classic notation 


% =======================================================
%  Example usage
% =======================================================

on rounded$
print_precision 5$

a := Phi((10.3-10)/0.3) - Phi((9.5-10)/0.3)$
b := Phi((10.5-10)/0.3) - Phi((9.5-10)/0.3)$
a/b;         % = 0.877

% alternativ we define Pr( a<=X<=b ) as normCDF2 

procedure normCDF2(a,b, mu,sigma); 
             1/2 * erf( (b-mu)/(sigma*sqrt(2)) ) -
             1/2 * erf( (a-mu)/(sigma*sqrt(2)) )$

Pab := normCDF2(9.5,10.3, 10, 0.3) / normCDF2(9.5,10.5, 10, 0.3);
