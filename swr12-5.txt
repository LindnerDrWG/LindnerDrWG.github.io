procedure VAR(lst);
begin
  scalar n, mu, ds;
  n := length(lst);  
  mu := (for each el in X sum el) / n;
  ds := for each el in X sum (el - mu)^2;
  return ds/n
end;

procedure VAR1(lst);
begin
  scalar n, mu, diff_sum;
  n := length(lst);
  if n = 0 then 0;  
  mu := for i := 1:length(lst) sum (part(lst, i)/n);
  diff_sum := for i := 1:n sum (part(lst, i) - mu)**2;
  return diff_sum / (n-1)
end;
