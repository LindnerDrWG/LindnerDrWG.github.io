% ------------------- helper functions ---------------------
COMMENT
   To calculate the t-TEST score 'T' we need the functions
   tCDF() and tINV() from t-distribution as helper, see Ch.3.3 
$

procedure tPDF(x,f);  
          1/(sqrt(f)*Beta(1/2,f/2))*(1+ x^2/f)^(-0.5*(f+1)) $
procedure tCDF(a,df);
   if a=0 then 0.5 else 
   if a<0 then 0.5 * ibeta(df/2, 0.5, df/(a^2 + df)) else
           1 - 0.5 * ibeta(df/2, 0.5, df/(a^2 + df));
procedure tinv(p,nu); 
% needs tcdf and tpdf
   begin scalar x, f, fx, dx,
      tol:= 1.0e-8, maxit:= 100;
      x := 0;  
      repeat <<
         f  := tcdf(x,nu) - p;
         fx := tpdf(x,nu);
         dx := f/fx;
         x := x - dx;
         maxit := maxit - 1
      >> until abs dx < tol or maxit = 0;
      return x
   end$
define dim = length$
procedure MEAN(X);  (for each el in X sum el)/length(X);
procedure SD(lst);
   begin
     scalar n, mu, ds;
     n := length(lst);  
     mu := (for each el in lst sum el) / n;
     ds := for each el in lst sum (el - mu)^2;
     return sqrt(ds/(n-1))
   end;


%#################################################
%  EXAMPLE: STUDENTS WEIGHT in Europe
%#################################################*/
% - a. -    We first do the t-test STEP-BY-STEP

on rounded $
print_precision 5$
X := {64.4, 68.5, 64.8, 58.9, 64.5, 68.6, 68.7, 62.9, 73.5}$

n  := dim(X)$
df := n - 1 ;            %-- degrees of freedom
mu0   := 70.8;           %-- the population mean */$
alpha := 0.05;           %-- set significance level */$

xbar := mean(X);
s := sd(X);

%-- calculate test statistic T */
T := (xbar - mu0) / (s/sqrt(n))   ;

%-- calculate critical value for lower.tail */
%critVal := tINV1(alpha, df) ;
critVal := tINV2(alpha, df) ;
    
%-- evaluate rejection region and decide about H0 */
if T > critVal then  write("reject H0") else  write("accept H0")$
done$
