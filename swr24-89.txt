procedure hgPDF(NN,RR,n,r); 
          binomial(RR,r)*binomial(NN-RR, n-r)/binomial(NN,n)$

procedure hgCDF(NN,RR,n, x);
          for k:=0:floor(x) sum hgPDF(NN,RR,n,k) $ 

procedure hgINV(alpha,NN,RR,n);
   % optimized by F.J. Wright
   begin scalar k := 0;
         repeat k := k+1 until hgCDF(NN,RR,n,k) >= alpha;
         return k
   end$
   
% -------------------- Example usage ---------------------

on rounded$
print_precision 5$

%     alpha N    R  n 
hgINV(0.99, 1000,10,50);   % --- = 3
