procedure chi2PDF(x, k);
          1/(2^(k/2)*gamma(k/2))*x^(k/2-1)*exp(-x/2.) $

% ------------------- Example usage ---------------------

on rounded $
print_precision 5$

for i:=1:9 do write {i*0.1, chi2PDF(i*0.1,6)};
