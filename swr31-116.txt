% ---------- inverse erf & normINV via Newton -------------
% a simple but stable version found by Gemini AI

precision 10;

procedure erfinv(y);
  begin scalar x, dx;
    x := y;              
    for i := 1:50 do 
	  << dx := (erf(x)-y)/(2/sqrt(pi) * exp(-x^2));
             x  := x - dx >>;
    return x
  end;

procedure normINV(q,m,s);
   begin  if q=0 then return "-Inf";
          if q=1 then return "+Inf";
          if 0<q and q<1 then 
             return m + sqrt(2) * s * erfinv(2*q-1)
    end$

procedure stdnormINV(q); normINV(q,0,1) $

% ---------- std.normal inverse via num_solve -----------

procedure normINV(q,m,s);
     % by F.J. Wright
     if q=0 then  infinity else
     if q=1 then -infinity else     
     if 0<q and q<1 then
     begin scalar y := 2*q-1, xx;
        if y = 0 then return 0;
        clear x;
        xx := num_solve(erf x = y, x = y); 
        xx := rhs first xx;
        return m + sqrt(2) * s * xx
    end$
