procedure weibullPDF(x,a,b);
          if (x<0) then 0.0 else
          b/a*(x/a)^(b-1)*exp(-(x/a)^b) $

% ------------------- Example usage ---------------------

on rounded $
print_precision 5$

shape := {1,2,3,4,5}$
for each i in shape do write {i, weibullPDF(3,2,i)};
