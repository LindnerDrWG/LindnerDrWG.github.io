           STATISTICS WITH REDUCE ch. 1.5
                      by W.G. Lindner 2026  $
                      
                      
procedure mad(lst);
begin
  scalar n, mu, diff_sum;
  n        := length(lst);  
  mu       := (for each el in lst sum el)/n;  % mean of list
  diff_sum := for each el in lst sum abs(el - mu);
  return diff_sum/n
end;

% ------------------- Example usage ---------------------

on rounded$
print_precision 5$

X := {4,36,45,50,75}$
mad(X);
