%|         bootstrapping for Confidence Inteval CI

% ---------------------- helper functions ------------------
procedure sum_(lst);   for each el in lst sum el $ 
procedure mean(lst);   sum_(lst)/length(lst) $ 
procedure listmake(m); for i:=1:m collect 0$
procedure percentile(X, perc);      
     part( sort(X,<), ceiling(length(X) * perc / 100));
define    dim = length $


%### -------------------------------------------------------
%###    bootstrapping for Confidence Inteval CI
%### -------------------------------------------------------

procedure bootci(X,IT);
begin scalar n, r, M, bootstrapMeans, estimatedMean;
      n := dim(X);           %-- number of samples
      M := listmake(n);     %-- container for all samples 
      bootstrapMeans := listmake(IT); 
      for j:=1:IT do         %-- perform bootstrap sampling
        begin           
          for i := 1:n do
              r := 1 + random(n); 
              M := (part(M,i) := part(X,r));
              bootstrapMeans  := (part(bootstrapMeans,j) := Mean(M))
            end;
       end;              
  estimatedMean := Mean(bootstrapMeans);
  ci2 := percentile(bootstrapMeans, 68.5);
  ci1 := percentile(bootstrapMeans,  2.5);
  write "est.Mean: ", estimatedMean;
  write "CI: ", {ci1,ci2}
END;
