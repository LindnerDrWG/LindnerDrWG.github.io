procedure percentile(X, perc);

%### ----------------------------------------------------------
%###    bootstrapping for Confidence Inteval CI
%### ----------------------------------------------------------

procedure bootci(X,IT);
begin scalar n, r, M, Mboot, estimatedMean;
      n := dim(X);           %-- number of samples
      M := listmake(IT);     %- container for all samples 
      Mboot := listmake(n);  %-- save a sample 
      for j:=1:IT do         %-- perform bootstrap sampling
        begin           
          for  i:=1:n do 
            begin r := 1 + random(n);
              Mboot := putNTH( Mboot, i, part(X,r));
              write "Mboot: ", Mboot;
              write "mean(Mboot) = ", Mean(Mboot);
              M     := putNTH( M, j, Mean(Mboot));
                              %-- bootstrap sample mean 
              write "M: ", M  
            end;
       end;                       %-------- mean(M) not reached --- ???
       estimatedMean := Mean(M);
       ci2 := percentile(M, 68.5);
       ci1 := percentile(M,  2.5);
       write "est.Mean: ", estimatedMean;
       write "CI: ", {ci1,ci2}
END;
