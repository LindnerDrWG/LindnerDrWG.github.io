% -------------- tiNV via num_solve ------- -----------------

procedure tINV(p,df); 
      <<
       clear x;
       if x=0.0 then -infinity else
       if x=1.0 then +infinity else
       rhs first num_solve(1-0.5*ibeta(df/2,0.5,df/(x^2+df))=p, x=2)
      >>$

% -------------- tiNV via NEWTON iteration  -----------------

procedure tPDF(x,f);  
          1/(sqrt(f)*Beta(1/2,f/2))*(1+ x^2/f)^(-0.5*(f+1)) $

procedure tCDF(a,df);
   if a=0 then 0.5 else 
   if a<0 then 0.5 * ibeta(df/2, 0.5, df/(a^2+df)) else
           1 - 0.5 * ibeta(df/2, 0.5, df/(a^2+df));

procedure tinv_NR(p,nu); 
   begin scalar x, f, fx, dx,
         tol:= 1.0e-8, maxit:= 100;
      x := 0;  
      repeat <<
         f  := tcdf(x,nu) - p;
         fx := tpdf(x,nu);
         dx := f/fx;
         x := x - dx;
         maxit := maxit - 1
      >> until abs dx < tol or maxit = 0;
      return x
   end$

% ----------------------- Example usage ----------------

on rounded $
print_precision 5$

%% ... wait ...

for df:=1:6 do write {df, tINV_NR(0.99,df)} ;
for df:=1:6 do write {df, tINV(0.99,df)} ;

$ ok
