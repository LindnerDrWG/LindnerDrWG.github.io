%  we choose the paramerization w.r.t. the urn model:
%               N  R  n r

procedure hgPDF(NN,RR,n,r); 
          binomial(RR,r)*binomial(NN-RR, n-r)/binomial(NN,n)$


COMMENT (EXCEL) ---------------------------------------------
    =HYPGEOM.DIST(x,n,k,m) of Excel is here named hgPDFexcel(x,n,k,m) 
    respecting the other ordering of the parameter sequence: 
    EXCEL> = HYPGEOM.DIST(2, 3, 4, 12, FALSE) + 
             HYPGEOM.DIST(3, 3, 4, 12, FALSE)
           = 0.218 + 0.18 = 0.236  
    --------------------------------------------------------- ;

procedure hgPDFexcel(x, n, k, m); 
          hgPDF(m,k,n,x)$ 

% -------------------- Example usage ---------------------

on rounded$
