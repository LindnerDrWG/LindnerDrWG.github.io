procedure fPDF(x,f1,f2);
          sqrt(((f1*x)^f1*f2^f2)/(f1*x+f2)^(f1+f2)) / (x * beta(f1/2,f2/2)) $

% -------------------- Example usage ---------------------

on rounded $
print_precision 5$

for x:=0:8 do if x=0 then write {0,0}
                  else write {x/2, fPDF(x/2, 10, 20)};
