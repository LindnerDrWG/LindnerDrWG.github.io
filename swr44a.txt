% ------------------- helper functions ---------------------
COMMENT
   To calculate the t-TEST score 'T' we need the functions
   tCDF() and tINV() from t-distribution as helper, see Ch.3.3 
$

procedure tPDF(x,f);  
          1/(sqrt(f)*Beta(1/2,f/2))*(1+ x^2/f)^(-0.5*(f+1)) $
procedure tCDF(a,df);
   if a=0 then 0.5 else 
   if a<0 then 0.5 * ibeta(df/2, 0.5, df/(a^2 + df)) else
           1 - 0.5 * ibeta(df/2, 0.5, df/(a^2 + df));
procedure tinv(p,nu); 
% needs tcdf and tpdf
   begin scalar x, f, fx, dx,
      tol:= 1.0e-8, maxit:= 100;
      x := 0;  
      repeat <<
         f  := tcdf(x,nu) - p;
         fx := tpdf(x,nu);
         dx := f/fx;
         x := x - dx;
         maxit := maxit - 1
      >> until abs dx < tol or maxit = 0;
      return x
   end$
procedure MEAN(X);  (for each el in X sum el)/length(X);
procedure SD(lst);
   begin
     scalar n, mu, ds;
     n := length(lst);  
     mu := (for each el in lst sum el) / n;
     ds := for each el in lst sum (el - mu)^2;
     return sqrt(ds/(n-1))       % with n-1, sample
   end;
define dim = length$


%#################################################
%      EXAMPLE: STUDENTS score
%#################################################*/
%      We first do the t-test-2 STEP-BY-STEP

on rounded $
print_precision 5$

A := {88, 92, 94, 78, 88, 95.}$
B := {75, 80, 79, 88, 85, 92.}$          

alpha := 0.05 $

n1 := dim(A) $
n2 := dim(B) $

m1 := mean(A) $
m2 := mean(B) $

s1 := sd(A) $
s2 := sd(B) $

%-- calculate the test statistic T */
s1 := 1/n1*s1*s1 ;
s2 := 1/n2*s2*s2 ;
df := (s1+s2)^2/(s1^2/(n1-1) + s2^2/(n2-1));   
%-- df = floor(f+0.5)     %- if you want integer */

T := (m1-m2)/sqrt(s1+s2) ;

%-- calculate critical value for lower.tail */
critVal := tINV(alpha/2, df) ;
    
%-- evaluate rejection region and decide about H0 */
if T > critVal then  write("reject H0") else  write("accept H0")$

%-------------------------------- method "both sided":

t95 := tINV(1-alpha/2, df);

faktor := sqrt(n1*n2/(n1+n2)) $
nenner := sqrt(((n1-1)*s1^2+(n2-1)*s2^2)/(n1+n2-2)) $

tquantil := tINV(1-alpha/2,df) ;

CI  := {mean(A)-mean(B)-tquantil*nenner/faktor,
       mean(A)-mean(B)+tquantil*nenner/faktor};


pValue := min( 2*tCDF(T,df), 2*(1-tCDF(T,df)) );
   
%-- evaluate rejection region and decide about H0 
if (T > critVal) then  write("reject H0") else  write("accept H0"); 
write("true difference in means is not equal to 0!");

COMMENT
-- DECISION
-- ERGO the alternative hypothesis is true: 
--      true difference in means is not equal to 0!
$
