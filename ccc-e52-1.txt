% 1. ---------- LHS of Stoke's formula (13.4) using avector
load_package avector;

F := avec(y, 4-x^2-y^2, x);         % substituted z = 4 - x^2 - y^2
S := avec(x, y, 4-x^2-y^2);

A := curl(F) dot (df(S,x) cross df(S,y)) ;

x := r*cos(theta);
y := r*sin(theta);

defint( defint( A * r, r, 0, 2), theta, 0, 2*pi);
