procedure SUM1(lst); 
          for each el in lst sum el $
             

procedure MEAN(lst); 
          sum1(lst)/length(lst)$

procedure DOT(A,B); 
          for i := 1:length(A) sum part(A,i)*part(B,i)$

procedure meanf(X,f); 
          dot(X,f)/Sum1(f)$

procedure sdF(X,f);
begin
  scalar n, muf, diff;
  n    := length(X);  
  if n = 0 then 0;  
  muf  := meanf(X,f);
  diff := for each el in X collect (el - muf)^2;
  return sqrt( dot(f, diff) / sum1(f))
end;

f :=  {5,18,42,27,8} $
