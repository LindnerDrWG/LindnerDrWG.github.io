%|     BOOTSTRAP for dependend samples with REDUCE

% ---------------------- helper functions ------------------
random_new_seed(1)$ % Seeding recommended before generating
procedure rnd();       random(10000000)/10000000.0 $
procedure randlist(n); for i:=1:n collect rnd() $
procedure sum_(lst);   for each el in lst sum el $ 
procedure mean(lst);   sum_(lst)/length(lst) $ 
procedure SD(lst);
begin scalar n, mu, ds, n:=length(lst);
  mu := (for each el in lst sum el)/n;  % mean
  ds := for each el in lst sum (el - mu)^2;
  return sqrt(ds/n)
end$
procedure listmake(m); for i:=1:m collect 0$
define    dim = length $


%-- boot1 determines an estimated mean resp. sd of a sample X.

%----------------------- boot1 --------------------------------
procedure boot1(X,IT);
%--------------------------------------------------------------
begin
  scalar n := length(X), r, M, Mboot, Sboot, estM, estS;
  M     := listmake(IT);   
  S     := listmake(IT);          
  Mboot := listmake(n);    
  Sboot := listmake(n);        
  for j:=1:IT do  
    begin                         %-- start bootstrap samples 
      for i:=1:n do               %-- create one resample of X
          begin r := 1 + random(n);  
            Mboot := (part(Mboot,i):=part(X,r));
            Sboot := (part(Sboot,i):=part(X,r));
           end;                    %-- bootstrap new sample 
        M := (part(M,j):=Mean(Mboot)); %-- j.th bootstrap mean 
        S := (part(S,j):=sd(Sboot));   %-- j.th bootstrap sd 
       write M(j)," ; ",S(j)       %-- show intermediate results
     end;                     
estM := Mean(M);                   %-- bootstrap sample mean/sd
estS := sd(S);                     %-- i.e. estimated values */ ;

 mat( (estMean_, estSD_),(estM, estS))
END;
% ----------------------- boot1 ---------------------------- END
