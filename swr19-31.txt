%-- We use the 'nearest-rank method' for calculating the quantile.

procedure quantile(X,p);              % improved by fjW
begin scalar Xs, i, dim;
      if p >= 100 then return max(X);
      if p <= 0 then return min(X);
      dim := length(X);
      Xs  := sort(X, <);
      i   := 1 + round((dim-1)*p/100);
      return part(Xs, i)
end;

define percentile = quantile;  %-- alias for Matlab compatibility
define Qp =         quantile;  %-- relax notation

% ------------------- Example usage ---------------------

X:={12, 3, 4, 56, 78, 18, 46, 78, 100};
% test
on rounded $
print_precision 5$

quantile(X, 0);
quantile(X, 25);
quantile(X, 50);
quantile(X, 75);
quantile(X, 100);

X:={12.0, 3.1, 4.2, 56.3, 78.4, 18.5, 46.6, 78.7, 100.8};

quantile(X, 0);
quantile(X, 25);
quantile(X, 50);
quantile(X, 75);
quantile(X, 100);

end;
