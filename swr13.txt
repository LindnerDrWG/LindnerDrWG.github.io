procedure SD(lst);
begin
  scalar n, mu, diff_sum;
  n := length(lst);  
  if n = 0 then 0;  
  mu := (for each el in lst sum el)/n;
  diff_sum := for each el in lst sum (el - mu)^2;
  return sqrt(diff_sum/n)
end;

   % The square root function can be input using the name sqrt, 
   % or the power operation ^(1/2)

procedure SD1(lst);
begin
  scalar n, mu, diff_sum;
  n := length(lst);
  if n = 0 then 0;  
  mu := (for each el in lst sum el)/n;
  ds := for each el in lst sum (el - mu)^2;
  return sqrt(ds / (n-1))
end;

% ------------------- Example usage ---------------------

on rounded$
print_precision 5$

X := {4, 36, 45, 50, 75};

on rounded;
print_precision 5;

sd(X);
sd1(X);
