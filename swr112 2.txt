procedure moment(X,r);
begin
  scalar n, mx, sx;
  n   := length(X);  
  mx := (for each el in X sum el)/n; % mean
  sx :=  for each el in X sum (el - mx)^r;
  return sx/n
end;


%--- To define kurtosis we only need Moment:

procedure kurtosis(X);
          moment(X,4)/moment(X,2)^2;   %--- compatibel with MatLab
    
% ------------------- Example usage ---------------------

on rounded $
print_precision 5 $

X := {4,36,45,50,75};
kurtosis(X); 

Y := {4, 5, 5, 6, 6, 6, 7, 8, 10, 12, 13};
