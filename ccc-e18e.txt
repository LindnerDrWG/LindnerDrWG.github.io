% permutation symbol as function
procedure Eijk(i,j,k);  (j-i)*(k-i)*(k-j)/2;
Eijk(1,2,3);
           
procedure cross(a,b); 
begin matrix c(1,3);
  for each i in {1,2,3} do 
    c(1,i):= for each j in {1,2,3} sum
             for each k in {1,2,3} sum 
                 Eijk(i,j, k) * a(1,j) * b(1,k);
    return c
  end;
