% ----- inverse erf and normINV by F.J. Wright------------

% Currently, erf x for |x| >= 5.0 does not evaluate numerically; see
% "alg/elem.red". It should evaluate to (approximately) sign x, so

let erf (~x) => sign x
    when numberp x and abs(x)>=5 and lisp !*rounded;

procedure numeric_erfinv(x);
   % Return a numerical approximation to the inverse error function
   % for real arguments x such that |x| < 1, outside which range it is
   % undefined.  Do this by solving erf y = x using Newton-Raphson
   % iteration.
   if abs x >= 1.0 then rederr "erfinv x defined only for |x| < 1"
   else if x = 0.0 then 0.0 else
   begin scalar const := sqrt(pi)/2.0,
                epsilon := 1.0e-12,
                dy, y := x;
      repeat <<
         dy := const * exp(y^2)*(erf(y) - x);
         y  := y - dy
      >> until abs dy < epsilon;
      return y
   end;

load_package specfn;
on rounded;
 
procedure normINV(q,m,s);
   if q=0 then -infinity else if q=1 then infinity else
   if 0<q and q<1 then 
          m + sqrt(2) * s * numeric_erfinv(2*q-1);

 
% -------------------- Example usage ---------------------

on rounded; precision 10;

normINV(0.8, 0,1);
normINV(0.6, 0,1);

normINV(0.80, 170, 5);
