load_package excalc;

% Define a 2D polar coordinate system with a metric
coframe o(r) = dr, o(phi) = r*dphi;

% Look at covariant (lower) vs contravariant (upper) indices
eps(r, phi);      % Returns 1
eps(-r, -phi);    % Returns 1 / r^2 
                  % adjusted automatically by the metric determinant


end;
