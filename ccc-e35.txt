% Number of pieces n and step size h
n := 100;
dx := 1.0/n;

% Initialise sum 
length := 0.0;

% Loop over the 100 segments
for i := 0 : n-1 do 
    <<  x1 := i*dx;      y1 := x1^2;
        x2 := (i+1)*dx;  y2 := x2^2;
        length := length + sqrt( (x2-x1)^2 + (y2-y1)^2 );
      >>;
length;

on rounded;
print_precision 5$
length;



on rounded;
print_precision 5;
n := 100;  h := 1.0/n; length := 0.0;
for i := 0 : n-1 sum 
     << x1 := i*h; x2 := (i+1)*h; hypot(x2-x1, x2^2-x1^2) >>;



load_package listvecops;
 
listproc labs v; sqrt(v *. v);   % list version of abs 
listproc g x; {x,x^2};
 
on rounded;
print_precision 5$
procedure h k; labs (g(k/100.0)-g((k-1)/100.0));
for k:=1:100 sum h(k);



end;
